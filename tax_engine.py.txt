# tax_engine.py - Upgraded calculation logic including METR and Scenarios

# 2025/2026 Estimated Saskatchewan & Federal Brackets
FED_BRACKETS = [(0, 57375, 0.15), (57375, 114750, 0.205), (114750, 177882, 0.26), (177882, 253414, 0.29), (253414, float('inf'), 0.33)]
SK_BRACKETS = [(0, 52057, 0.105), (52057, 148734, 0.125), (148734, float('inf'), 0.15)]

FED_BPA = 16183
SK_BPA = 18491

# Disability Tax Credit (DTC)
FED_DTC = 9686
FED_DTC_SUPP = 5655
SK_DTC = 10090
SK_DTC_SUPP = 10090

def calculate_marginal_tax(taxable_income, brackets):
    tax = 0.0
    for low, high, rate in brackets:
        if taxable_income > low:
            tax += (min(taxable_income, high) - low) * rate
    return tax

def calculate_individual_tax(income, rrsp, med, childcare, has_dtc, is_child=False):
    # Determine Net Income
    net_income = max(0, income - rrsp - childcare)
    
    # Calculate Gross Taxes
    fed_tax_gross = calculate_marginal_tax(net_income, FED_BRACKETS)
    sk_tax_gross = calculate_marginal_tax(net_income, SK_BRACKETS)
    
    # Non-Refundable Credits (BPA, DTC, Med)
    fed_credits = FED_BPA
    sk_credits = SK_BPA
    
    if has_dtc:
        fed_credits += FED_DTC + (FED_DTC_SUPP if is_child else 0)
        sk_credits += SK_DTC + (SK_DTC_SUPP if is_child else 0)
        
    med_threshold = min(0.03 * net_income, 2759)
    if med > med_threshold:
        fed_credits += (med - med_threshold)
        sk_credits += (med - med_threshold) # SK uses similar threshold mechanics
        
    fed_nrtc = fed_credits * 0.15
    sk_nrtc = sk_credits * 0.105
    
    # Net Taxes
    net_fed = max(0, fed_tax_gross - fed_nrtc)
    net_sk = max(0, sk_tax_gross - sk_nrtc)
    
    return net_fed + net_sk, net_income

def optimize_family_taxes(s1_inc, s1_rrsp, s1_withheld, s2_inc, s2_rrsp, s2_withheld, total_med, total_childcare, c2_has_dtc):
    """ Optimizes deductions (Medical/Childcare) between spouses. """
    s1_net_est = s1_inc - s1_rrsp
    s2_net_est = s2_inc - s2_rrsp
    
    # Childcare generally must be claimed by lower income spouse. 
    # Limit is $11k for DTC child + $8k for non-DTC child = $19,000 max.
    childcare_limit = 11000 if c2_has_dtc else 8000
    childcare_limit += 8000 # Add child 1
    allowed_childcare = min(total_childcare, childcare_limit)
    
    s1_cc = allowed_childcare if s1_net_est < s2_net_est else 0
    s2_cc = allowed_childcare if s2_net_est <= s1_net_est else 0
    
    # Medical is also usually best on lower net income due to the 3% threshold
    s1_med = total_med if s1_net_est < s2_net_est else 0
    s2_med = total_med if s2_net_est <= s1_net_est else 0

    s1_tax, s1_net = calculate_individual_tax(s1_inc, s1_rrsp, s1_med, s1_cc, False)
    s2_tax, s2_net = calculate_individual_tax(s2_inc, s2_rrsp, s2_med, s2_cc, False)

    total_tax = s1_tax + s2_tax
    total_refund = (s1_withheld - s1_tax) + (s2_withheld - s2_tax)
    
    return {
        "spouse1_tax": s1_tax,
        "spouse2_tax": s2_tax,
        "total_tax": total_tax,
        "total_refund": total_refund,
        "s1_cc": s1_cc,
        "s2_cc": s2_cc,
        "s1_med": s1_med,
        "s2_med": s2_med
    }

def analyze_rrsp_scenario(s1_inc, s1_rrsp, s2_inc, s2_rrsp, additional_rrsp, apply_to_s1=True):
    """ Models the exact tax savings of contributing $X more to an RRSP. """
    base = optimize_family_taxes(s1_inc, s1_rrsp, 0, s2_inc, s2_rrsp, 0, 0, 0, False)
    
    if apply_to_s1:
        scenario = optimize_family_taxes(s1_inc, s1_rrsp + additional_rrsp, 0, s2_inc, s2_rrsp, 0, 0, 0, False)
    else:
        scenario = optimize_family_taxes(s1_inc, s1_rrsp, 0, s2_inc, s2_rrsp + additional_rrsp, 0, 0, 0, False)
        
    tax_savings = base['total_tax'] - scenario['total_tax']
    return tax_savings
